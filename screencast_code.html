Acquiring the Data from the API

if (file.exists("json_response.json")) {
  
  json_response <- read_json("json_response.json")
  
} else { 
  uastring <- 
    "R/python data science demos, youremail@email.com"
  
  request <- 
    request(
      "https://clinicaltrials.gov/api/v2/studies?query.cond=cluster+headache&aggFilters=results%3Awith%2Cstatus%3Acom"
    ) %>% 
    req_perform()
  
  json_response <- resp_body_json(request)
  
  # saving json_response if it didn't exist in the first place
  jsonlite::write_json(json_response, "json_response.json")
}

Cleaning the groups

group_table <- 
  first_study_results %>% 
  separate_longer_delim(
    cols = group,
    delim = ";"
  ) %>% 
  mutate(
    group = str_trim(group),
    group_number = 
      ceiling(row_number()/3)
  ) %>% 
  group_by(
    group_number
  ) %>% 
  summarise(
    group_id = first(group),
    group_title = nth(group, 2),
    group_description = nth(group, 3)
  ) %>% 
  select(
    -group_number
  ) %>% 
  distinct()

Cleaning the Classes

class_table <- 
  first_study_results %>% 
  separate_longer_delim(
    cols = class,
    delim = ";"
  ) %>% 
  mutate(
    class = str_trim(class),
    class_title =
      if_else(
        str_detect(class, "[^A-Z0-9]"),
        if_else(
          str_detect(class, "0-9\\.0-9"),
          NA,
          class
        ),
        NA
      ), 
    class_title = 
      if_else(
        str_detect(
          class_title, "[:digit:]+\\.[:digit:]?"  
        ),
        NA,
        class_title
      )
  ) %>% 
  group_by(title) %>% 
  fill(class_title, .direction = "down") %>% 
  mutate(
    class_title = 
      if_else(
        is.na(class_title),
        title,
        class_title
      )
  ) %>% 
  filter(
    !class == class_title
  ) %>% 
  mutate(
    group_number = 
      if_else(
        is.na(dispersion_type),
        ceiling(row_number()/2),
        ceiling(row_number()/3)
      )
  ) %>% 
  group_by(type, title, group_number, description, population_description, reporting_status,param_type,unit_of_measure,time_frame,dispersion_type, class_title) %>% 
  summarise(
    group_id = first(class),
    value = nth(class, 2),
    spread = 
      if_else(
        is.na(nth(class,3)),
        NA,
        nth(class,3)
      )
  ) %>% 
  ungroup() %>% 
  select(-group_number)
`summarise()` has grouped output by 'type', 'title', 'group_number',
'description', 'population_description', 'reporting_status', 'param_type',
'unit_of_measure', 'time_frame', 'dispersion_type'. You can override using the
`.groups` argument.

Data Manipulaiton for Better Graphs

full_results <- 
  full_results %>% 
  mutate(
    time = 
      case_when(
        str_detect(class_title, "after 2 weeks") ~ "After 2 Weeks",
        str_detect(class_title, "at 2 weeks") ~ "After 2 Weeks",
        str_detect(class_title, "at 30 min.") ~ "Pain Free at 30 minutes",
        str_detect(class_title, "at 15 minutes") ~ "Pain Free at 15 minutes",
        T ~ "Baseline"
      ),
    time = factor(time, levels = c("Baseline", "After 2 Weeks", "Pain Free at 15 minutes", "Pain Free at 30 minutes")),
    event = 
      str_remove(class_title, "at baseline| after 2 weeks|at 15 minutes|at 30 min.|at 2 weeks"),
    event = 
      if_else(
        str_detect(event, "Number of"),
        "Number of Attacks Treated",
        event
      ),
    event = 
      case_when(
        str_detect(event, "Activity baseline") ~ "Activity",
        T ~ event
      ),
    event = str_trim(event)
  )
